### tool_list.php
```php
<?php
// P2.1 spike, evidence 5: the agent's offered tool list, name by name, on v3 or v4.
// Run: php ../../concrete/bin/concrete c5:exec <this file>
// Prints: name <TAB> sha1(description) <TAB> sha1(properties schema + required) <TAB> class, sorted by name; then the count.

use Katalysis\NeuronAi\ConcreteCmsAgent;
use NeuronAI\Tools\ToolInterface;
use NeuronAI\Tools\Toolkits\ToolkitInterface;

$agent = new ConcreteCmsAgent();
$rows = [];
foreach ($agent->getTools() as $item) {
    $tools = $item instanceof ToolkitInterface ? $item->tools() : [$item];
    foreach ($tools as $tool) {
        if (!$tool instanceof ToolInterface) {
            continue;
        }
        $props = [];
        foreach ($tool->getProperties() as $p) {
            $props[$p->getName()] = $p->getJsonSchema();
        }
        ksort($props);
        $required = $tool->getRequiredProperties();
        sort($required);
        $rows[$tool->getName()] = implode("\t", [
            $tool->getName(),
            sha1((string) $tool->getDescription()),
            sha1(json_encode([$props, $required])),
            $tool::class,
        ]);
    }
}
ksort($rows);
echo implode("\n", $rows), "\n";
echo 'count ', count($rows), "\n";

return 0;
```

<?php
// P2.1 spike: which src/ files extend or implement a Neuron class that the installed Neuron lacks.
// Such a file fatals when loaded, and Doctrine's annotation driver loads every file under src/.
// Run from the package folder: php <this file>
$it = new RecursiveIteratorIterator(new RecursiveDirectoryIterator('src'));
foreach ($it as $f) {
    if ($f->getExtension() !== 'php') {
        continue;
    }
    $code = file_get_contents($f->getPathname());
    preg_match_all('/^use\s+([^;\s]+)(?:\s+as\s+(\w+))?;/m', $code, $uses, PREG_SET_ORDER);
    $map = [];
    foreach ($uses as $u) {
        $alias = $u[2] ?? '' ?: substr(strrchr('\\' . $u[1], '\\'), 1);
        $map[$alias] = ltrim($u[1], '\\');
    }
    preg_match_all('/\b(?:extends|implements)\s+([\\\\\w]+(?:\s*,\s*[\\\\\w]+)*)/', $code, $m);
    foreach ($m[1] as $list) {
        foreach (preg_split('/\s*,\s*/', $list) as $name) {
            $fq = str_starts_with($name, '\\') ? ltrim($name, '\\') : ($map[strtok($name, '\\')] ?? null);
            if ($fq !== null && str_contains($name, '\\') && !str_starts_with($name, '\\')) {
                $fq .= substr($name, strpos($name, '\\'));
            }
            if ($fq === null || !str_starts_with($fq, 'NeuronAI\\')) {
                continue;
            }
            $file = 'vendor/neuron-core/neuron-ai/src/' . str_replace('\\', '/', substr($fq, 9)) . '.php';
            echo (is_file($file) ? 'ok      ' : 'MISSING ') . str_pad($f->getPathname(), 48) . " $fq\n";
        }
    }
}
